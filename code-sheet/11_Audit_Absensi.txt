function buatSheetAuditAbsensi(){

  var ss =
    SpreadsheetApp.getActiveSpreadsheet();


  var sheet =
    ss.getSheetByName(
      SHEET_AUDIT_ABSENSI
    );


  if(!sheet){

    sheet =
      ss.insertSheet(
        SHEET_AUDIT_ABSENSI
      );


    sheet.appendRow([

      "Waktu_Audit",
      "Jenis_Audit",
      "Tanggal",
      "ID",
      "Nama",
      "Baris_Data",
      "Detail",
      "Status"

    ]);

  }


  return sheet;

}

function auditKualitasAbsensiHarian(){

  var ss =
    SpreadsheetApp.getActiveSpreadsheet();


  var absensiSheet =
    ambilSheet(
      ss,
      SHEET_ABSENSI_HARIAN
    );


  var auditSheet =
    ambilSheet(
      ss,
      SHEET_AUDIT_ABSENSI
    );


  var data =
    absensiSheet
    .getDataRange()
    .getValues();


  var sekarang =
    new Date();


  var hasilAudit = [];


  for(
    var i = 1;
    i < data.length;
    i++
  ){

    var row =
      data[i];


    var baris =
      i + 1;


    var tanggal =
      row[
        COL_ABSENSI.TANGGAL - 1
      ];


    var id =
      row[
        COL_ABSENSI.ID - 1
      ];


    var nama =
      row[
        COL_ABSENSI.NAMA - 1
      ];


    var jamMasuk =
      row[
        COL_ABSENSI.JAM_MASUK - 1
      ];


    var jamPulang =
      row[
        COL_ABSENSI.JAM_PULANG - 1
      ];


    var statusAbsen =
      row[
        COL_ABSENSI.STATUS_ABSEN - 1
      ];


    var keterangan =
      row[
        COL_ABSENSI.KETERANGAN - 1
      ];


    var jamKerja =
      Number(
        row[
          COL_ABSENSI.JAM_KERJA - 1
        ]
        ||
        0
      );


    var lembur =
      Number(
        row[
          COL_ABSENSI.LEMBUR - 1
        ]
        ||
        0
      );


    var jamKurang =
      Number(
        row[
          COL_ABSENSI.JAM_KERJA_KURANG - 1
        ]
        ||
        0
      );


    /*
     * CEK 1
     * ID kosong
     */

    if(!id){

      hasilAudit.push(
        buatBarisAuditAbsensi(
          sekarang,
          "ID Kosong",
          tanggal,
          id,
          nama,
          baris,
          "Data absensi tidak memiliki ID",
          "Perlu Dicek"
        )
      );

    }



    /*
     * CEK 2
     * Jam Pulang tanpa Jam Masuk
     */

    if(
      jamPulang
      &&
      !jamMasuk
    ){

      hasilAudit.push(
        buatBarisAuditAbsensi(
          sekarang,
          "Jam Pulang Tanpa Masuk",
          tanggal,
          id,
          nama,
          baris,
          "Jam pulang tersedia tetapi jam masuk kosong",
          "Perlu Dicek"
        )
      );

    }



    /*
     * CEK 3
     * Status Lengkap tapi jam pulang kosong
     */

    if(
      statusAbsen == "Lengkap"
      &&
      !jamPulang
    ){

      hasilAudit.push(
        buatBarisAuditAbsensi(
          sekarang,
          "Status Tidak Konsisten",
          tanggal,
          id,
          nama,
          baris,
          "Status Lengkap tetapi jam pulang kosong",
          "Perlu Dicek"
        )
      );

    }



    /*
     * CEK 4
     * Lembur tanpa keterangan
     */

    if(
      lembur > 0
      &&
      keterangan != "Pulang Lembur"
    ){

      hasilAudit.push(
        buatBarisAuditAbsensi(
          sekarang,
          "Lembur Tidak Sesuai",
          tanggal,
          id,
          nama,
          baris,
          "Ada lembur tetapi keterangan bukan Pulang Lembur",
          "Perlu Dicek"
        )
      );

    }



    /*
     * CEK 5
     * Jam kurang tanpa pulang awal
     */

    if(
      jamKurang > 0
      &&
      keterangan != "Pulang Lebih Awal"
    ){

      hasilAudit.push(
        buatBarisAuditAbsensi(
          sekarang,
          "Jam Kerja Kurang Tidak Sesuai",
          tanggal,
          id,
          nama,
          baris,
          "Jam kerja kurang tetapi keterangan bukan Pulang Lebih Awal",
          "Perlu Dicek"
        )
      );

    }


  }


  if(
    hasilAudit.length > 0
  ){

    auditSheet
    .getRange(
      auditSheet.getLastRow()+1,
      1,
      hasilAudit.length,
      8
    )
    .setValues(
      hasilAudit
    );

  }


  SpreadsheetApp.getUi().alert(
    "Audit kualitas absensi selesai.\n\n" +
    "Temuan: " +
    hasilAudit.length
  );


}

function buatBarisAuditAbsensi(
  waktu,
  jenis,
  tanggal,
  id,
  nama,
  baris,
  detail,
  status
){

  return [

    waktu,

    jenis,

    tanggal,

    id,

    nama,

    baris,

    detail,

    status

  ];

}
