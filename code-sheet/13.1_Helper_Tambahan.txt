/**
 * Menghitung jam kerja normal dalam satuan MENIT
 * Rumus Koreksi: (jamPulang - jamMasuk) - istirahat + batasMasuk
 */
function kalkulasiJamKerjaNormalMenit(jamMasuk, jamPulang, istirahatMenit, batasMasukMenit) {
  var menitMasuk = ubahJamKeMenit(jamMasuk);
  var menitPulang = ubahJamKeMenit(jamPulang);

  if (menitMasuk === null || menitPulang === null) {
    return 0;
  }

  // Penanganan Shift Malam (jika jam pulang melewati tengah malam)
  if (menitPulang < menitMasuk) {
    menitPulang += 1440; // Tambah 24 jam (1440 menit)
  }

  var istirahat = Number(istirahatMenit || 0);
  var batasMasuk = Number(batasMasukMenit || 0);

  var totalMenit = (menitPulang - menitMasuk) - istirahat + batasMasuk;

  return totalMenit > 0 ? totalMenit : 0;
}

/**
 * Memperbarui sel kolom Jam Kerja Normal di sheet SETTING (Kolom P, mulai baris 11)
 */
function updateSemuaJamKerjaNormalSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var settingSheet = ambilSheet(ss, "SETTING");
  
  var dataShift = ambilSemuaDataShift();

  for (var i = 0; i < dataShift.length; i++) {
    var kodeShift = dataShift[i][0];
    
    if (!kodeShift) continue;

    var jamMasuk = dataShift[i][2];
    var jamPulang = dataShift[i][3];
    var batasMasuk = Number(dataShift[i][5] || 0);
    var istirahat = Number(dataShift[i][8] || 0);

    var hasilMenit = kalkulasiJamKerjaNormalMenit(jamMasuk, jamPulang, istirahat, batasMasuk);

    settingSheet.getRange(11 + i, 16).setValue(hasilMenit);
  }
}
