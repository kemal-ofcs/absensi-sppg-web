/****************************************************
 * 03_Operator_Login.gs
 * LOGIN OPERATOR SCANNER
 * --------------------------------------------------
 * Fungsi utama:
 * - Menerima username & password dari frontend scanner.
 * - Mengautentikasi operator scanner dari SHEET_OPERATOR_SCANNER.
 * - Mengembalikan status login dan identitas operator.
 ****************************************************/

function loginScanner(e) {
  var parameter = e && e.parameter ? e.parameter : {};
  var callback = bersihkanCallback(parameter.callback);
  var username = String(parameter.username || "").trim();
  var password = String(parameter.password || "").trim();

  if (!username || !password) {
    return buatJsonpResponse(callback, {
      sukses: false,
      status: "Gagal",
      pesan: "Username dan password wajib diisi."
    });
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var operatorSheet = ambilSheet(ss, SHEET_OPERATOR_SCANNER);
    var dataOperator = operatorSheet.getDataRange().getValues();

    for (var i = 1; i < dataOperator.length; i++) {
      var rowNumber = i + 1;
      var kodeOperator = String(dataOperator[i][0] || "").trim();
      var uName = String(dataOperator[i][1] || "").trim();
      var pWord = String(dataOperator[i][2] || "").trim();
      var namaOperator = String(dataOperator[i][3] || "").trim();
      var status = String(dataOperator[i][4] || "").trim();
      var role = String(dataOperator[i][5] || "").trim();

      if (uName.toLowerCase() === username.toLowerCase() && pWord === password) {
        if (status !== "Aktif") {
          return buatJsonpResponse(callback, {
            sukses: false,
            status: "Gagal",
            pesan: "Akun operator nonaktif. Silakan hubungi admin."
          });
        }

        // Update Waktu Login Terakhir di Kolom G (7) dengan format rapi
        operatorSheet
          .getRange(rowNumber, 7)
          .setValue(new Date())
          .setNumberFormat("dd/MM/yyyy HH:mm:ss");

        return buatJsonpResponse(callback, {
          sukses: true,
          status: "Berhasil",
          pesan: "Login operator berhasil.",
          operator: {
            kodeOperator: kodeOperator,
            username: uName,
            nama: namaOperator,
            role: role
          }
        });
      }
    }

    return buatJsonpResponse(callback, {
      sukses: false,
      status: "Gagal",
      pesan: "Username atau password salah."
    });

  } catch (error) {
    return buatJsonpResponse(callback, {
      sukses: false,
      status: "Error",
      pesan: "Terjadi kesalahan di server: " + error.message
    });
  }
}
