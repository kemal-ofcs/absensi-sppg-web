function prosesKoreksiAdmin() {

  var ss =
    SpreadsheetApp.getActiveSpreadsheet();


  var koreksiSheet =
    ambilSheet(
      ss,
      SHEET_KOREKSI_ADMIN
    );


  var absensiSheet =
    ambilSheet(
      ss,
      SHEET_ABSENSI_HARIAN
    );


  var masterSheet =
    ambilSheet(
      ss,
      SHEET_MASTER_DATA
    );


  var koreksiData =
    koreksiSheet
    .getDataRange()
    .getValues();


  var masterData =
    masterSheet
    .getDataRange()
    .getValues();

  var operatorSheet =
  ambilSheet(
    ss,
    SHEET_OPERATOR_SCANNER
  );


var operatorData =
  operatorSheet
  .getDataRange()
  .getValues();



  for(
    var i = 1;
    i < koreksiData.length;
    i++
  ){


    var rowNumber =
      i + 1;


    var tanggal =
      koreksiData[i][1];


    var id =
      koreksiData[i][2];


    var nama =
      koreksiData[i][3];


    var divisi =
      koreksiData[i][4];


    var jenisKoreksi =
      koreksiData[i][5];


    var jamKoreksi =
      koreksiData[i][6];


    var keteranganAdmin =
      koreksiData[i][7];


    var statusProses =
      koreksiData[i][8];

    var kodeOperator =
      koreksiData[i][11];

    if(
      statusProses !== "Belum Diproses"
    ){
      continue;
    }



    if(
      !tanggal ||
      !id ||
      !jenisKoreksi
    ){

      koreksiSheet
      .getRange(rowNumber,9)
      .setValue("Gagal");


      koreksiSheet
      .getRange(rowNumber,10)
      .setValue(new Date());


      continue;

    }



    var dataMaster =
      cariDataMaster(
        masterData,
        id
      );



    if(
      !dataMaster.ditemukan ||
      dataMaster.statusAktif !== "Aktif"
    ){

      koreksiSheet
      .getRange(rowNumber,9)
      .setValue("Gagal");


      koreksiSheet
      .getRange(rowNumber,10)
      .setValue(new Date());


      continue;

    }



    if(!nama){

      nama =
      dataMaster.nama;

      koreksiSheet
      .getRange(rowNumber,4)
      .setValue(nama);

    }



    if(!divisi){

      divisi =
      dataMaster.kelasDivisi;

      koreksiSheet
      .getRange(rowNumber,5)
      .setValue(divisi);

    }

    var operator =
    cariOperator(
      operatorData,
      kodeOperator
    );


    if(
    !operator.ditemukan ||
    operator.status != "Aktif"
    ){

      koreksiSheet
      .getRange(rowNumber,9)
      .setValue("Gagal");


      koreksiSheet
      .getRange(rowNumber,10)
      .setValue(new Date());


      koreksiSheet
      .getRange(rowNumber,8)
      .setValue(
        "Kode Operator tidak valid"
      );


      continue;

    }

    var idReferensi =
    generateIdReferensiKoreksi(
      koreksiSheet
    );


    koreksiSheet
    .getRange(
      rowNumber,
      11
    )
    .setValue(
      idReferensi
    );

    var berhasil =
    updateAbsensiDariKoreksi(
      absensiSheet,
      tanggal,
      id,
      nama,
      divisi,
      jenisKoreksi,
      jamKoreksi,
      keteranganAdmin,
      idReferensi,
      kodeOperator
    );



    koreksiSheet
    .getRange(rowNumber,9)
    .setValue(
      berhasil
      ?
      "Sudah Diproses"
      :
      "Gagal"
    );


    koreksiSheet
    .getRange(rowNumber,10)
    .setValue(
      new Date()
    );


  }

}

/****************************************************
 * KOREKSI TERLAMBAT
 ****************************************************/


if(
  jenisKoreksi == "Terlambat"
){

  var hasilMasuk = {

    menitTerlambat:0,

    menitDatangAwal:0

  };

  var waktuMasukKoreksi =
    gabungkanTanggalDanJam(
      tanggal,
      jamKoreksi
    );



  var dataShift =
    bacaShiftDariTblShift(
      shift
    );



  if(
    !dataShift.ditemukan
  ){

    return false;

  }



  var jamMasukShift =
    ubahJamKeMenit(
      dataShift.jamMasukShift
    );



  var batasNormalMasuk =
    jamMasukShift
    +
    Number(
      dataShift.batasMasukMenit || 0
    );



  var batasAkhirMasuk =
    batasNormalMasuk
    +
    Number(
      dataShift.toleransiMasukMenit || 0
    );



  var menitMasuk =
    waktuMasukKoreksi.getHours()
    *
    60
    +
    waktuMasukKoreksi.getMinutes();



  var menitTerlambat = 0;

  var menitDatangAwal = 0;



  /*
   * DATANG AWAL
   */

  if(
    menitMasuk < jamMasukShift
  ){

    menitDatangAwal =
      jamMasukShift
      -
      menitMasuk;

  }



  /*
   * TERLAMBAT
   *
   * dihitung setelah batas normal masuk
   */

  if(
    menitMasuk > batasNormalMasuk
  ){

    menitTerlambat =
      menitMasuk
      -
      batasNormalMasuk;

  }



  hasilMasuk.menitTerlambat =
    menitTerlambat;


  hasilMasuk.menitDatangAwal =
    menitDatangAwal;



  Logger.log({

    DEBUG_WAKTU_MASUK:
      waktuMasukKoreksi,

    DEBUG_SHIFT:
      shift,

    DEBUG_JAM_MASUK_SHIFT:
      jamMasukShift,

    DEBUG_BATAS_NORMAL_MASUK:
      batasNormalMasuk,

    DEBUG_BATAS_AKHIR_MASUK:
      batasAkhirMasuk,

    DEBUG_MENIT_MASUK:
      menitMasuk,

    DEBUG_TERLAMBAT:
      menitTerlambat,

    DEBUG_DATANG_AWAL:
      menitDatangAwal

  });



  /*
   * UPDATE ABSENSI
   */


  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.JAM_MASUK
  )
  .setValue(
    waktuMasukKoreksi
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.STATUS_KEHADIRAN
  )
  .setValue(
    "Hadir"
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.MENIT_TERLAMBAT
  )
  .setValue(
    menitTerlambat
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.MENIT_DATANG_AWAL
  )
  .setValue(
    menitDatangAwal
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.KETERANGAN
  )
  .setValue(
    keteranganAdmin ||
    "Koreksi Admin - Terlambat"
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.SUMBER
  )
  .setValue(
    SUMBER_KOREKSI_ADMIN
  );



  absensiSheet
  .getRange(
    barisDitemukan,
    COL_ABSENSI.UPDATE_TERAKHIR
  )
  .setValue(
    sekarang
  );



  /*
   * LOG SCAN KOREKSI
   */


  tambahLogKoreksiAdmin(

    logSheet,

    tanggal,

    jamKoreksi,

    id,

    nama,

    divisi,

    "Masuk",

    "Berhasil",

    "Koreksi Admin - Terlambat",

    keteranganAdmin ||
    "Terlambat",

    idReferensi,

    kodeOperator

  );



  return true;


}

absensiSheet

absensiSheet

);

absensiSheet
.getRange(
  barisDitemukan,
  COL_ABSENSI.JAM_KERJA
)
.setValue(0);

.getRange(
  barisDitemukan,

}

function cekPrioritasKoreksiAdmin(
  absensiSheet,
  tanggal,
  id,
  jenisScan
) {

  var dataAbsensi =
    absensiSheet
    .getDataRange()
    .getValues();


  for(
    var i = 1;
    i < dataAbsensi.length;
    i++
  ){

    var tanggalAbsensi =
      dataAbsensi[i][
        COL_ABSENSI.TANGGAL - 1
      ];


    var idAbsensi =
      dataAbsensi[i][
        COL_ABSENSI.ID - 1
      ];


    if(
      idAbsensi == id &&
      tanggalSama(
        tanggalAbsensi,
        tanggal
      )
    ){

      var statusKehadiran =
        dataAbsensi[i]
        [
          COL_ABSENSI.STATUS_KEHADIRAN - 1
        ];


      var statusAbsen =
        dataAbsensi[i]
        [
          COL_ABSENSI.STATUS_ABSEN - 1
        ];


      var sumberData =
        dataAbsensi[i]
        [
          COL_ABSENSI.SUMBER - 1
        ];


      var keterangan =
        dataAbsensi[i]
        [
          COL_ABSENSI.KETERANGAN - 1
        ];



      /*
       * Koreksi Admin yang mengunci:
       *
       * Sakit
       * Izin
       * Dispen
       * Alfa
       *
       * Karena status absen = Tidak Hadir
       */

      if(
        sumberData == SUMBER_KOREKSI_ADMIN &&
        statusAbsen == "Tidak Hadir"
      ){

        return {

          terkunci:true,

          pesan:

          "Data absensi sudah dikoreksi admin." +

          (
            statusKehadiran
            ?
            " Status: " + statusKehadiran
            :
            ""
          )

          +

          (
            keterangan
            ?
            ". Keterangan: " + keterangan
            :
            ""

          )

        };

      }



      return {

        terkunci:false,

        pesan:""

      };


    }

  }



  return {

    terkunci:false,

    pesan:""

  };

}

function apiAdminCorrection(e) {
  var parameter = e && e.parameter ? e.parameter : {};
  var callback = bersihkanCallback(parameter.callback);
  
  var hasil = {
    sukses: false,
    status: "Gagal",
    pesan: "Gagal memproses koreksi."
  };

  try {
    // Ambil parameter dari frontend
    var tanggal = parameter.tanggal;
    var idKaryawan = parameter.id;
    var jenisKoreksi = parameter.jenis_koreksi;
    var jamKoreksi = parameter.jam_koreksi;
    var keterangan = parameter.keterangan;
    var kodeOperator = parameter.kode_operator;

    if (!tanggal || !idKaryawan || !jenisKoreksi) {
      hasil.pesan = "Tanggal, ID Karyawan, dan Jenis Koreksi wajib diisi.";
      return buatJsonpResponse(callback, hasil);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var koreksiSheet = ambilSheet(ss, SHEET_KOREKSI_ADMIN);

    // Masukkan baris baru ke sheet koreksi admin secara otomatis
    koreksiSheet.appendRow([
      "", // Kolom 1 (Timestamp/ID otomatis jika ada)
      new Date(tanggal),
      idKaryawan,
      "", // Nama (nanti dilengkapi proses)
      "", // Divisi
      jenisKoreksi,
      jamKoreksi || "",
      keterangan || "Koreksi via Web App",
      "Belum Diproses",
      "",
      "",
      kodeOperator || ""
    ]);

    // Jalankan fungsi proses koreksi utama
    prosesKoreksiAdmin();

    hasil = {
      sukses: true,
      status: "Berhasil",
      pesan: "Data koreksi berhasil diajukan dan diproses sistem."
    };

    return buatJsonpResponse(callback, hasil);

  } catch (error) {
    hasil = {
      sukses: false,
      status: "Error",
      pesan: "Terjadi error: " + error.message
    };
    return buatJsonpResponse(callback, hasil);
  }
}
