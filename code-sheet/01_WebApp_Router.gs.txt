/****************************************************
 * 01_WebApp_Router.gs
 * ROUTING WEB APP
 * --------------------------------------------------
 * Fungsi utama:
 * - Mengatur pintu masuk Web App.
 * - Menerima request login scanner.
 * - Menerima request API scanner v1.3.
 * - Menampilkan halaman scanner jika mode scanner.
 * - Menolak jalur lama /exec?id=ID&token=TOKEN.
 ****************************************************/

function doGet(e) {

  var parameter = e && e.parameter
    ? e.parameter
    : {};

  var mode =
    String(parameter.mode || "")
    .trim()
    .toLowerCase();

  var action =
    String(parameter.action || "")
    .trim()
    .toLowerCase();


  if(action === "login_scanner") {
    return loginScanner(e);
  }

  if(action === "scanner_api") {
    return apiScannerV13(e);
  }

  if(action === "admin_correction") {
    return apiAdminCorrection(e);
  }

  if(action === "penugasan_backup") {
    return apiPenugasanBackup(e);
  }

  if(action === "batal_backup") {
    return apiBatalBackup(e);
  }

  if(action === "audit_kualitas") {
    return apiAuditKualitas(e);
  }

  if(action === "import_offline") {
    return apiImportOffline(e);
  }

  if(action === "idcard_by_id") {
    return apiIdCardById(e);
  }

  if(action === "idcard_massal") {
    return apiIdCardMassal(e);
  }

  if(action === "export_qr_png") {
    return apiExportQrPng(e);
  }

  if(action === "get_data_karyawan") {
    return apiGetDataKaryawan(e);
  }

  if(action === "arsip_import_offline"){
    return apiArsipImportOffline(e);
  }

  if(mode === "scanner") {
    return HtmlService
      .createHtmlOutputFromFile("Scanner")
      .setTitle(
        "Secure Scanner Mode v1.3"
      );
  }

  return ContentService
    .createTextOutput(
      JSON.stringify({
        sukses:false,
        status:"Ditolak",
        pesan: "Akses tidak valid."
      })
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );

}

/****************************************************
 * API HANDLER DATA KARYAWAN
 ****************************************************/

function apiGetDataKaryawan(e) {
  var callback = e && e.parameter ? e.parameter.callback : "";
  var hasil = getDataKaryawanMaster();
  return buatJsonpResponse(callback, hasil);
}

function getDataKaryawanMaster() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ambilSheet(ss, SHEET_MASTER_DATA);

    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return { sukses: true, data: [] };
    }

    var rawData = sheet.getRange(2, 1, lastRow - 1, 20).getValues();

    var formattedData = rawData.map(function(row) {
      return row.map(function(cell) {
        if (cell instanceof Date) {
          return Utilities.formatDate(cell, ss.getSpreadsheetTimeZone(), "yyyy-MM-dd HH:mm:ss");
        }
        return cell;
      });
    });

    return {
      sukses: true,
      data: formattedData
    };
  } catch (err) {
    return {
      sukses: false,
      pesan: "Gagal mengambil data karyawan: " + err.toString()
    };
  }
}

/****************************************************
 * API HANDLER AUDIT KUALITAS ABSENSI
 ****************************************************/

function apiAuditKualitas(e) {
  var callback = e && e.parameter ? e.parameter.callback : "";
  var hasil = jalankanDanAmbilAuditKualitas();
  return buatJsonpResponse(callback, hasil);
}

function jalankanDanAmbilAuditKualitas() {
  try {
    var ringkasan = generateAlfaHarian();
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ambilSheet(ss, SHEET_AUDIT_ABSENSI);
    var logs = [];
    
    if (sheet && sheet.getLastRow() > 1) {
      var lastRow = sheet.getLastRow();
      var startRow = Math.max(2, lastRow - 49);
      var numRows = lastRow - startRow + 1;
      var rawData = sheet.getRange(startRow, 1, numRows, 8).getValues();
      
      var tz = ss.getSpreadsheetTimeZone();
      logs = rawData.map(function(row) {
        return {
          waktu: row[0] instanceof Date ? Utilities.formatDate(row[0], tz, "yyyy-MM-dd HH:mm:ss") : String(row[0] || "-"),
          jenis: String(row[1] || "-"),
          tanggal: row[2] instanceof Date ? Utilities.formatDate(row[2], tz, "yyyy-MM-dd") : String(row[2] || "-"),
          id: String(row[3] || "-"),
          nama: String(row[4] || "-"),
          baris: String(row[5] || "-"),
          detail: String(row[6] || "-"),
          status: String(row[7] || "-")
        };
      }).reverse();
    }
    
    return {
      sukses: true,
      ringkasan: ringkasan,
      logs: logs,
      pesan: "Audit Kualitas Absensi berhasil dijalankan."
    };
  } catch (err) {
    return {
      sukses: false,
      pesan: "Gagal menjalankan audit: " + err.toString()
    };
  }
}

/****************************************************
 * API HANDLER GENERATE ID CARD BY ID
 ****************************************************/
function apiIdCardById(e) {
  var parameter = e && e.parameter ? e.parameter : {};
  var callback = bersihkanCallback(parameter.callback);
  var rawIds = String(parameter.id || parameter.ids || "").trim();

  if (!rawIds) {
    return buatJsonpResponse(callback, {
      sukses: false,
      status: "Gagal",
      pesan: "ID karyawan wajib diisi."
    });
  }

  try {
    var daftarId = rawIds
      .split(",")
      .map(function(id) { return String(id || "").trim(); })
      .filter(function(id) { return id !== ""; });

    var hasilAmbil = massPrintAmbilSiswaByDaftarId(daftarId);

    if (hasilAmbil.daftarSiswa.length === 0) {
      return buatJsonpResponse(callback, {
        sukses: false,
        status: "Gagal",
        pesan: "Tidak ada karyawan aktif/valid dari ID yang diinput."
      });
    }

    var hasil = massPrintBuatPdfBatchSiswa(hasilAmbil.daftarSiswa, "By ID via Web App");

    massPrintUpdateStatusMasterBatch(
      hasilAmbil.daftarSiswa,
      "Berhasil",
      hasil.pdfUrl,
      "Berhasil dibuat via Web App API."
    );

    massPrintCatatLog(
      "By ID via Web App",
      hasilAmbil.daftarSiswa.length,
      hasil.jumlahBatch,
      hasil.jumlahSlide,
      hasil.pdfUrl,
      "Berhasil",
      "Generate ID Card by ID berhasil."
    );

    return buatJsonpResponse(callback, {
      sukses: true,
      status: "Berhasil",
      pesan: "Generate ID Card berhasil dibuat untuk " + hasilAmbil.daftarSiswa.length + " karyawan.",
      pdfUrl: hasil.pdfUrl,
      jumlahSiswa: hasilAmbil.daftarSiswa.length
    });

  } catch (err) {
    return buatJsonpResponse(callback, {
      sukses: false,
      status: "Error",
      pesan: "Gagal generate ID Card: " + err.message
    });
  }
}

function exportQrPngEngineWeb_() {
  try {
    exportQRCodePNGKeDrive();
  } catch(e) {
    if (e.message && e.message.indexOf("getUi") !== -1) {
      Logger.log("Abaikan alert UI di Web App context.");
    } else {
      throw e;
    }
  }
}
