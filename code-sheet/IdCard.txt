/**
 * Membuat payload QR aman.
 * Format:
 * ID_Unik|Token_Absensi
 */
function idCardBuatPayloadQr(siswa) {
  if (!siswa || !siswa.id || !siswa.tokenAbsensi) {
    throw new Error("Data siswa tidak lengkap untuk membuat payload QR.");
  }

  return siswa.id + "|" + siswa.tokenAbsensi;
}

 * Membuat QR Code dari payload.
 */
/**
 * Membuat QR Code dari payload.
 *
 * Cache berdasarkan payload:
 * ID_Unik|Token_Absensi
 *
 * Jika token berubah:
 * payload berubah
 * cache otomatis berbeda
 */
function idCardBuatQrBlob(payloadQr) {

  if (!payloadQr) {
    throw new Error(
      "Payload QR kosong."
    );
  }


  var cache =
    CacheService.getScriptCache();


  /**
   * Key memakai payload lengkap.
   *
   * Contoh:
   * USR001|ABC123
   *
   * Bukan hanya:
   * USR001
   */
  var cacheKey =
    "QR_" +
    Utilities
      .base64EncodeWebSafe(payloadQr);


  var cached =
    cache.get(cacheKey);


  if (cached) {

    var cachedBlob =
      Utilities.newBlob(
        Utilities.base64Decode(cached),
        "image/png"
      );


    cachedBlob.setName(
      "qr-" +
      payloadQr.split("|")[0] +
      ".png"
    );


    return cachedBlob;
  }


  var url =
    "https://quickchart.io/qr" +
    "?text=" +
    encodeURIComponent(payloadQr) +
    "&size=350" +
    "&margin=2";


  var response =
    UrlFetchApp.fetch(
      url,
      {
        muteHttpExceptions:true
      }
    );


  var code =
    response.getResponseCode();


  if (code !== 200) {

    throw new Error(
      "Gagal membuat QR. HTTP Code: " +
      code
    );

  }


  var blob =
    response.getBlob();


  blob.setName(
    "qr-" +
    payloadQr.split("|")[0] +
    ".png"
  );


  /**
   * Simpan cache 6 jam.
   *
   * Cukup untuk satu sesi generate.
   */
  cache.put(
    cacheKey,
    Utilities.base64Encode(
      blob.getBytes()
    ),
    21600
  );


  return blob;
}

/**
 * Export Google Slides menjadi PDF.
 */
function idCardExportPdf(slidesFileId, namaPdf, outputFolder) {
  var slidesFile = DriveApp.getFileById(slidesFileId);
  var pdfBlob = slidesFile.getAs(MimeType.PDF);

  pdfBlob.setName(namaPdf);

  var pdfFile = outputFolder.createFile(pdfBlob);

  return pdfFile;
}

function exportQRCodePNGKeDrive() {

  const SHEET_NAME = "QR_CODE";
  const FOLDER_ID = IDCARD_OUTPUT_FOLDER_PNG;


  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAME);
  const folder = DriveApp.getFolderById(FOLDER_ID);


  if (!sh) {
    throw new Error("Sheet QR_CODE tidak ditemukan.");
  }


  // Mengikuti jumlah ID_Unik
  const idData =
    sh.getRange("A:A")
      .getValues()
      .filter(String);


  const lastRow = idData.length;


  if (lastRow <= 1) {
    Logger.log("Tidak ada data QR.");
    return;
  }



  // Ambil A - I
  const data =
    sh.getRange(2,1,lastRow-1,9)
      .getValues();



  let berhasil = 0;
  let dilewati = 0;
  let gagal = 0;



  data.forEach((row,index)=>{


    const nomorBaris = index + 2;


    const idUnik = row[0];
    const nama = row[1];
    const payloadQR = row[3];

    const statusQR = row[5];



    // Skip jika sudah export
    if(statusQR === "Sudah Export"){

      dilewati++;

      return;

    }



    if(!idUnik || !nama || !payloadQR){

      dilewati++;

      return;

    }



    try {


      const qrUrl =
        "https://quickchart.io/qr?text=" +
        encodeURIComponent(payloadQR);



      const response =
        UrlFetchApp.fetch(qrUrl);



      const namaFile =
        buatNamaFileAman(
          idUnik + "_" + nama
        )
        + ".png";



      const blob =
        response.getBlob()
          .setName(namaFile)
          .setContentType("image/png");



      const file =
        folder.createFile(blob);



      // Update status
      sh.getRange(nomorBaris,6)
        .setValue("Sudah Export");



      // Update tanggal
      sh.getRange(nomorBaris,8)
        .setValue(new Date());



      // Update link PNG
      sh.getRange(nomorBaris,9)
        .setValue(file.getUrl());



      berhasil++;


    }
    catch(err){


      gagal++;

      Logger.log(
        "Gagal baris "
        + nomorBaris
        + " : "
        + err.message
      );


    }


  });



  SpreadsheetApp.getUi()
    .alert(
      "Export QR selesai.\n\n" +
      "Berhasil : " + berhasil + "\n" +
      "Dilewati : " + dilewati + "\n" +
      "Gagal : " + gagal
    );


}


function buatNamaFileAman(nama) {
  return String(nama)
    .replace(/[\\\/:*?"<>|#\[\]]/g, "_")
    .replace(/\s+/g, "_")
    .trim();
}


function tentukanNamaFile(folder, namaDasar, ekstensi) {
  const namaUtama = namaDasar + "." + ekstensi;

  // Kalau belum ada, pakai nama utama
  if (!folder.getFilesByName(namaUtama).hasNext()) {
    return namaUtama;
  }

  // Kalau sudah ada, buat nama duplikat dengan timestamp
  const timestamp = Utilities.formatDate(
    new Date(),
    Session.getScriptTimeZone(),
    "yyyyMMdd_HHmmss"
  );

  return namaDasar + "__duplikat_" + timestamp + "." + ekstensi;
}
