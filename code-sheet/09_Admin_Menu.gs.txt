function onOpen() {

  ui.createMenu("Admin Absensi")
    // ==========================
    // ABSENSI
    // ==========================
    .addItem("1. Generate Token Absensi", "generateTokenAbsensi")
    .addItem("2. Proses Koreksi Admin", "prosesKoreksiAdmin")
    .addItem("3. Proses Penugasan Baris Backup", "prosesPenugasanBackupBarisAktif")
    .addItem("4. Proses Pembatalan Baris Backup", "batalkanPenugasanBackupBarisAktif")
    .addItem("5. Proses Import Offline", "prosesImportOffline")
    .addItem("6. Arsip Import Offline", "arsipImportOffline")

    .addSeparator()

    // ==========================
    // AUDIT & SISTEM
    // ==========================
    .addItem("7. Audit Kualitas Absensi", "jalankanAuditAbsensi")
    .addItem("8. Setup Trigger Generate Alfa", "setupTriggerGenerateAlfa")
    .addItem("9. Refresh Dashboard", "refreshDashboard")

    .addSeparator()

    // ==========================
    // ID CARD & QR
    // ==========================
    .addItem("10. Generate ID Card by ID", "massPrintGenerateByInputId")
    .addItem("11. Generate ID Card Semua Aktif/Belum Berhasil", "massPrintGenerateSemuaAktif")
    .addItem("12. Export QR Code PNG ke Drive", "exportQRCodePNGKeDrive")

    .addToUi();
}

/****************************************************
 * TRIGGER OTOMATIS ONEDIT TERPADU
 ****************************************************/
function onEdit(e) {
  if (!e || !e.range) return;

  var sheet = e.range.getSheet();
  var namaSheet = sheet.getName();
  var baris = e.range.getRow();
  var kolom = e.range.getColumn();

  // 1. OTOMATISASI SHEET "SETTING"
  if (namaSheet === "SETTING") {
    if (baris >= 11 && (kolom === 11 || kolom === 12 || kolom === 14 || kolom === 17)) {
      updateSemuaJamKerjaNormalSheet();
    }
    return;
  }

  // 2. CONTROL PANEL ADMIN (CHECKBOX)
  var nilai = e.range.getValue();

  if (nilai !== true) {
    return;
  }

  var cell = e.range.getA1Notation();
  var proses = null;

  if (namaSheet === SHEET_MASTER_DATA && cell === "S1") {
    proses = "generateTokenAbsensi";
  }
  else if (namaSheet === SHEET_KOREKSI_ADMIN && cell === "O1") {
    proses = "prosesKoreksiAdmin";
  }
  else if (namaSheet === SHEET_QR_CODE && cell === "L1") {
    proses = "massPrintGenerateByInputId";
  }
  else if (namaSheet === SHEET_QR_CODE && cell === "L2") {
    proses = "massPrintGenerateSemuaAktif";
  }
  else if (namaSheet === SHEET_QR_CODE && cell === "L3") {
    proses = "exportQRCodePNGKeDrive";
  }
  else if ((namaSheet === "BACKUP_KARYAWAN" || (typeof SHEET_BACKUP_KARYAWAN !== "undefined" && namaSheet === SHEET_BACKUP_KARYAWAN)) && cell === "W1") {
    proses = "prosesPenugasanBackupBarisAktif";
  }
  else if ((namaSheet === "BACKUP_KARYAWAN" || (typeof SHEET_BACKUP_KARYAWAN !== "undefined" && namaSheet === SHEET_BACKUP_KARYAWAN)) && cell === "W2") {
    proses = "batalkanPenugasanBackupBarisAktif";
  }
  else if (namaSheet === SHEET_IMPORT_OFFLINE && cell === "O1") {
    proses = "prosesImportOffline";
  }
  else if (namaSheet === SHEET_IMPORT_OFFLINE && cell === "O2") {
    proses = "arsipImportOffline";
  }
  else if (namaSheet === SHEET_AUDIT_ABSENSI && cell === "K1") {
    proses = "jalankanAuditAbsensi";
  }
  else if (namaSheet === SHEET_AUDIT_ABSENSI && cell === "K2") {
    proses = "setupTriggerGenerateAlfa";
  }
  else if (namaSheet === SHEET_DASHBOARD && cell === "C1") {
    proses = "refreshDashboard";
  }

  if (!proses) {
    e.range.setValue(false);
    return;
  }

  try {
    SpreadsheetApp.getActive().toast("Menjalankan: " + proses, "Admin Absensi", 5);

    if (typeof this[proses] === "function") {
      this[proses]();
    } else {
      throw new Error("Fungsi " + proses + " tidak ditemukan.");
    }

    SpreadsheetApp.getActive().toast("Selesai: " + proses, "Admin Absensi", 5);
  } catch (error) {
    SpreadsheetApp.getActive().toast("Error: " + error.message, "Admin Absensi", 8);
  } finally {
    e.range.setValue(false);
  }
}

function kontrolPanelTrigger(e) {
  onEdit(e);
}
