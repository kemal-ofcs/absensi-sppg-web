/****************************************************
 * 14_Scan_Settings.gs
 * SETTING JAM SCAN ABSENSI
 * --------------------------------------------------
 * Fungsi utama:
 * - Membaca aturan scan berdasarkan SHIFT.
 * - Mengatur batas datang awal.
 * - Mengatur keterlambatan.
 * - Mengatur jam pulang.
 *
 * Sumber utama:
 * tblShift pada sheet SETTING
 ****************************************************/


function bacaSettingAbsensi() {

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var settingSheet = ambilSheet(ss, "SETTING");

  var dataSetting = settingSheet
    .getRange("I2:J20")
    .getValues();

  var setting = {};


  for (var i = 0; i < dataSetting.length; i++) {

    var namaSetting =
      String(dataSetting[i][0] || "").trim();

    var nilaiSetting =
      dataSetting[i][1];


    if (namaSetting) {
      setting[namaSetting] = nilaiSetting;
    }

  }


  return {

    // dipertahankan sementara
    // untuk kompatibilitas sistem lama

    jamBukaMasuk:
      setting["JAM_BUKA_MASUK"],

    jamTutupMasuk:
      setting["JAM_TUTUP_MASUK"],

    jamBukaPulang:
      setting["JAM_BUKA_PULANG"],

    toleransiMasuk:
      setting["TOLERANSI_MASUK"],

    jamKerjaNormal:
      setting["JAM_KERJA_NORMAL"],

    istirahatKerja:
      setting["ISTIRAHAT_KERJA"]

  };

}



/****************************************************
 * KONVERSI JAM KE MENIT
 ****************************************************/

function ubahJamKeMenit(nilaiJam) {


  if (!nilaiJam) {
    return null;
  }


  if (
    Object.prototype.toString.call(nilaiJam)
    === "[object Date]"
    &&
    !isNaN(nilaiJam)
  ) {

    var jam =
      Number(
        Utilities.formatDate(
          nilaiJam,
          Session.getScriptTimeZone(),
          "HH"
        )
      );


    var menit =
      Number(
        Utilities.formatDate(
          nilaiJam,
          Session.getScriptTimeZone(),
          "mm"
        )
      );


    return jam * 60 + menit;

  }


  var teksJam =
    String(nilaiJam).trim();


  var bagian =
    teksJam.split(":");


  if (bagian.length < 2) {
    return null;
  }


  var jam =
    parseInt(bagian[0],10);

  var menit =
    parseInt(bagian[1],10);


  if (
    isNaN(jam)
    ||
    isNaN(menit)
  ) {

    return null;

  }


  return jam * 60 + menit;

}

/****************************************************
 * ATURAN SCAN ONLINE BERDASARKAN SHIFT
 ****************************************************/
function cekAturanJamScanOnline(
  sekarang,
  sudahMasuk,
  sudahPulang,
  waktuMasuk,
  scanTerakhir,
  jenisScanTerakhir,
  nama,
  id,
  shift
) {

  // ==============================
  // SHIFT FLEKSIBEL
  // ==============================
  if (cekShiftFleksibel(shift)) {

    if (!sudahMasuk && !sudahPulang) {
      return {
        boleh: true,
        jenisScan: "Masuk",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Fleksibel",
        menitTerlambat: 0,
        menitDatangAwal: 0
      };
    }

    if (sudahMasuk && !sudahPulang) {
      return {
        boleh: true,
        jenisScan: "Pulang",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Fleksibel",
        menitTerlambat: 0,
        menitDatangAwal: 0
      };
    }

    return {
      boleh: false,
      jenisScan: "Pulang Ditolak",
      statusProses: "Ditolak",
      catatanSistem: "Karyawan shift fleksibel sudah melakukan scan pulang",
      pesanBrowser:
        "Gagal: " +
        nama +
        " (" +
        id +
        ") sudah melakukan absen masuk dan pulang."
    };
  }

  // ==============================
  // VALIDASI SHIFT
  // ==============================
  if (!shift) {
    return {
      boleh: false,
      jenisScan: "Masuk Ditolak",
      statusProses: "Ditolak",
      catatanSistem: "Shift karyawan belum diatur",
      pesanBrowser:
        "Absensi ditolak. Shift belum diatur. Silakan hubungi operator."
    };
  }

  var dataShift = bacaShiftDariTblShift(shift);

  if (!dataShift.ditemukan) {
    return {
      boleh: false,
      jenisScan: "Masuk Ditolak",
      statusProses: "Ditolak",
      catatanSistem:
        "Kode shift " +
        shift +
        " tidak ditemukan pada tblShift",
      pesanBrowser:
        "Absensi ditolak. Konfigurasi shift tidak valid. Silakan hubungi operator."
    };
  }

  var totalMenit =
    sekarang.getHours() * 60 +
    sekarang.getMinutes();

  var jamMasukShift =
    ubahJamKeMenit(dataShift.jamMasukShift);

  var jamPulangShift =
    ubahJamKeMenit(dataShift.jamPulangShift);

  if (
    jamMasukShift === null ||
    jamPulangShift === null
  ) {
    return {
      boleh: false,
      jenisScan: "Setting Shift Error",
      statusProses: "Ditolak",
      catatanSistem: "Konfigurasi jam shift tidak valid",
      pesanBrowser:
        "Konfigurasi shift tidak valid. Silakan hubungi operator."
    };
  }

  var batasAwalDatang =
    jamMasukShift -
    Number(dataShift.batasDatangAwalMenit || 0);

  var batasNormalMasuk =
    jamMasukShift +
    Number(dataShift.batasMasukMenit || 0);

  var batasAkhirMasuk =
    batasNormalMasuk +
    Number(dataShift.toleransiMasukMenit || 0);

  var batasPulangMenit =
    Number(dataShift.batasPulangMenit || 0);

  // Riwayat Pulang Perlu Verifikasi juga masuk ke sini.
  // Jangan izinkan scan berikutnya menjadi Masuk/Pulang lagi.
  if (sudahPulang) {
    return {
      boleh: false,
      jenisScan: "Pulang Ditolak",
      statusProses: "Ditolak",
      catatanSistem: "Scan pulang sudah tercatat sebelumnya",
      pesanBrowser:
        "Gagal: " +
        nama +
        " (" +
        id +
        ") sudah melakukan scan pulang untuk tanggal absensi ini."
    };
  }

  // ==============================
  // PROSES TANPA RIWAYAT MASUK
  // ==============================
  if (!sudahMasuk) {

    /*
     * Hitung jarak maju dari jam pulang.
     * Rumus ini tetap benar jika jendela pulang
     * melewati pukul 00.00.
     */
    var menitSejakJamPulang =
      totalMenit - jamPulangShift;

    if (menitSejakJamPulang < 0) {
      menitSejakJamPulang += 1440;
    }

    if (
      menitSejakJamPulang >= 0 &&
      menitSejakJamPulang <= batasPulangMenit
    ) {
      return {
        boleh: true,
        jenisScan: "Pulang",
        statusProses: "Perlu Verifikasi",
        statusKehadiran: "Hadir",
        keterangan: "Perlu Verifikasi",
        menitTerlambat: 0,
        menitDatangAwal: 0,
        catatanSistem:
          "Scan pulang tanpa data scan masuk",
        pesanBrowser:
          "Jam Pulang dicatat dengan status Perlu Verifikasi.\n\n" +
          "Nama: " +
          nama +
          " (" +
          id +
          ")\n\n" +
          "Data jam masuk tidak ditemukan. Silakan hubungi operator."
      };
    }

    // ==============================
    // PROSES MASUK
    // ==============================

    // Terlalu awal.
    if (totalMenit < batasAwalDatang) {
      return {
        boleh: false,
        jenisScan: "Masuk Ditolak - Terlalu Awal",
        statusProses: "Ditolak",
        catatanSistem:
          "Scan sebelum batas datang awal shift",
        pesanBrowser:
          "Absensi belum dibuka. Shift " +
          dataShift.namaShift +
          " mulai menerima scan pukul " +
          formatJamKey(
            new Date(
              1899,
              11,
              30,
              Math.floor(batasAwalDatang / 60),
              batasAwalDatang % 60
            )
          )
      };
    }

    // Datang lebih awal.
    if (totalMenit < jamMasukShift) {
      var menitDatangAwal =
        jamMasukShift - totalMenit;

      return {
        boleh: true,
        jenisScan: "Masuk",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Datang Lebih Awal",
        menitTerlambat: 0,
        menitDatangAwal: menitDatangAwal
      };
    }

    // Tepat waktu.
    if (totalMenit <= batasNormalMasuk) {
      return {
        boleh: true,
        jenisScan: "Masuk",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Tepat Waktu",
        menitTerlambat: 0,
        menitDatangAwal: 0
      };
    }

    // Terlambat dalam toleransi.
    if (totalMenit <= batasAkhirMasuk) {
      var menitTerlambat =
        totalMenit - batasNormalMasuk;

      return {
        boleh: true,
        jenisScan: "Masuk",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Terlambat",
        menitTerlambat: menitTerlambat,
        menitDatangAwal: 0
      };
    }

    return {
      boleh: false,
      jenisScan: "Masuk Ditolak",
      statusProses: "Ditolak",
      catatanSistem:
        "Melewati batas toleransi masuk",
      pesanBrowser:
        "Waktu absensi masuk sudah ditutup. Silakan hubungi operator."
    };
  }

  // ==============================
  // PROSES PULANG SETELAH MASUK
  // ==============================
  if (sudahMasuk && !sudahPulang) {

    var batasMultiScan =
      Number(
        bacaSettingGlobal(
          "BATAS_MULTI_SCAN_MENIT"
        ) || 0
      );

    if (
      scanTerakhir &&
      jenisScanTerakhir == "Masuk" &&
      batasMultiScan > 0
    ) {
      var selisihMenit =
        (
          sekarang.getTime() -
          scanTerakhir.getTime()
        ) / 60000;

      if (selisihMenit <= batasMultiScan) {
        return {
          boleh: false,
          jenisScan: "Multi Scan Ditolak",
          statusProses: "Ditolak",
          catatanSistem:
            "Kemungkinan scan masuk ganda. Selisih scan " +
            Math.floor(selisihMenit) +
            " menit.",
          pesanBrowser:
            "Scan ditolak. Kemungkinan Anda melakukan scan masuk ulang."
        };
      }
    }

    /*
     * Ubah jam shift malam menjadi garis waktu berurutan.
     * Contoh shift 20.00-05.00:
     * 20.00 = 1200 dan 05.00 hari berikutnya = 1740.
     */
    var totalMenitPulang =
      totalMenit;

    var jamPulangPembanding =
      jamPulangShift;

    if (jamPulangShift < jamMasukShift) {
      jamPulangPembanding += 1440;

      if (
        waktuMasuk instanceof Date &&
        !isNaN(waktuMasuk.getTime())
      ) {
        var tanggalMasukObj =
          new Date(
            waktuMasuk.getFullYear(),
            waktuMasuk.getMonth(),
            waktuMasuk.getDate()
          );

        var tanggalSekarangObj =
          new Date(
            sekarang.getFullYear(),
            sekarang.getMonth(),
            sekarang.getDate()
          );

        var selisihHari =
          Math.round(
            (
              tanggalSekarangObj.getTime() -
              tanggalMasukObj.getTime()
            ) / 86400000
          );

        if (selisihHari > 0) {
          totalMenitPulang +=
            selisihHari * 1440;
        }
      } else if (
        totalMenit < jamMasukShift
      ) {
        totalMenitPulang += 1440;
      }
    }

    var batasAkhirPulang =
      jamPulangPembanding +
      batasPulangMenit;

    if (totalMenitPulang > batasAkhirPulang) {
      return {
        boleh: false,
        jenisScan: "Pulang Ditolak",
        statusProses: "Ditolak",
        catatanSistem:
          "Melewati batas waktu pulang shift",
        pesanBrowser:
          "Scan ditolak. Batas waktu pulang shift sudah berakhir. Silakan hubungi operator."
      };
    }

    if (totalMenitPulang < jamPulangPembanding) {
      return {
        boleh: true,
        jenisScan: "Pulang",
        statusProses: "Berhasil",
        statusKehadiran: "Hadir",
        keterangan: "Pulang Lebih Awal",
        catatanSistem: "Pulang Lebih Awal",
        pesanBrowser:
          "Jam Pulang berhasil dicatat. Status: Pulang Lebih Awal."
      };
    }

    return {
      boleh: true,
      jenisScan: "Pulang",
      statusProses: "Berhasil",
      statusKehadiran: "Hadir",
      keterangan: "Pulang Normal"
    };
  }

  return {
    boleh: false,
    jenisScan: "Ditolak",
    statusProses: "Ditolak",
    catatanSistem: "Kondisi riwayat scan tidak valid",
    pesanBrowser:
      "Gagal: kondisi riwayat absensi tidak valid. Silakan hubungi operator."
  };
}

/****************************************************
 * TENTUKAN TANGGAL ABSENSI BERDASARKAN SHIFT
 * --------------------------------------------------
 * Fungsi:
 * - Menentukan tanggal kerja absensi.
 * - Menangani shift yang melewati tengah malam.
 *
 * Contoh:
 *
 * Shift Malam:
 * Masuk 22/07 22:00
 * Pulang 23/07 07:00
 *
 * Maka tanggal absensi:
 * 22/07
 ****************************************************/

function tentukanTanggalAbsensi(sekarang, shift) {



  var tanggalAbsensi =
    new Date(
      sekarang.getFullYear(),
      sekarang.getMonth(),
      sekarang.getDate()
    );


  /*
  * Shift fleksibel menggunakan tanggal kalender.
  * Tidak perlu membaca tblShift.
  */
  if (
    !shift ||
    cekShiftFleksibel(shift)
  ) {
    return tanggalAbsensi;
  }


  var dataShift =
  bacaShiftDariTblShift(shift);


Logger.log(
  "SHIFT: " + shift
);


Logger.log(
  JSON.stringify(dataShift)
);


if (!dataShift.ditemukan) {
  return tanggalAbsensi;
}


var jamMasuk =
  ubahJamKeMenit(
    dataShift.jamMasukShift
  );


var jamPulang =
  ubahJamKeMenit(
    dataShift.jamPulangShift
  );


var menitSekarang =
  sekarang.getHours() * 60
  +
  sekarang.getMinutes();


var batasPulangShift =
  jamPulang +
  Number(
    dataShift.batasPulangMenit || 0
  );


Logger.log(
  "jamMasuk: " + jamMasuk
);


Logger.log(
  "jamPulang: " + jamPulang
);


Logger.log(
  "menitSekarang: " + menitSekarang
);


Logger.log(
  "batasPulangShift: " + batasPulangShift
);


Logger.log(
  "shift malam: " +
  (jamPulang < jamMasuk)
);

  /*
   * SHIFT LEWAT TENGAH MALAM
   *
   * Contoh:
   *
   * Masuk 22:00
   * Pulang 07:00
   *
   */

  if (jamPulang < jamMasuk) {


    /*
     * Jika scan terjadi sebelum jam pulang,
     * berarti masih bagian shift kemarin.
     *
     * Contoh:
     *
     * 23/07 05:00
     *
     * tetap dihitung:
     *
     * 22/07
     */

    /*
    * SHIFT MALAM
    *
    * Digunakan untuk menentukan tanggal absensi.
    * Tidak sama dengan batas validasi pulang.
    *
    * Tujuan:
    * - Tetap mencari histori shift malam.
    * - Menghindari scan pulang dianggap scan masuk baru.
    *
    * Buffer digunakan untuk kasus:
    * - lupa scan pulang
    * - keterlambatan scan
    * - koreksi admin
    */

    var batasDeteksiShiftMalam =
      jamPulang
      +
      Number(dataShift.batasPulangMenit || 0)
      +
      Number(
        dataShift.bufferDeteksiShiftMalamMenit || 0
      );


    if (
      menitSekarang <= batasDeteksiShiftMalam
    ) {

      tanggalAbsensi.setDate(
        tanggalAbsensi.getDate() - 1
      );

    }

  }



  return tanggalAbsensi;

}
