function prosesImportOffline() {
  var lock = LockService.getDocumentLock();

  if (!lock.tryLock(30000)) {
    throw new Error(
      "Sistem sedang memproses data lain. Silakan coba kembali."
    );
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    prosesImportOfflineEngine(
      ambilSheet(ss, SHEET_IMPORT_OFFLINE),
      ambilSheet(ss, SHEET_ABSENSI_HARIAN),
      ambilSheet(ss, SHEET_MASTER_DATA),
      ambilSheet(ss, SHEET_LOG_SCAN)
    );
  } finally {
    lock.releaseLock();
  }
}

function prosesImportOfflineEngine(
  importSheet,
  absensiSheet,
  masterSheet,
  logSheet
) {
  var importData = importSheet.getDataRange().getValues();
  var masterData = masterSheet.getDataRange().getValues();

  var jumlahBerhasil = 0;
  var jumlahGagal = 0;
  var jumlahLewat = 0;

  for (var i = 1; i < importData.length; i++) {
    var row = i + 1;

    var data = {
      timestamp: importData[i][0],
      tanggal: importData[i][1],
      id: importData[i][2],
      nama: importData[i][3],
      divisi: importData[i][4],
      jamMasuk: importData[i][5],
      jamPulang: importData[i][6],
      statusKehadiran: importData[i][7],
      statusAbsen: importData[i][8],
      keterangan: importData[i][9],
      statusProses: importData[i][10]
    };

    try {
      var statusImport = validasiStatusImport(data.statusProses);

      if (!statusImport.valid) {
        tandaiImportGagalFinal(importSheet, row, statusImport.pesan);
        jumlahGagal++;
        continue;
      }

      if (statusImport.aksi == "lewati") {
        jumlahLewat++;
        continue;
      }

      var validasi = validasiDataImportFinal(data);

      if (!validasi.valid) {
        tandaiImportGagalFinal(importSheet, row, validasi.pesan);
        jumlahGagal++;
        continue;
      }

      var tanggalNormal = validasi.tanggal;
      var user = cariDataUserDiMaster(masterData, data.id);

      if (!user.ditemukan) {
        tandaiImportGagalFinal(importSheet, row, "ID tidak ditemukan di MASTER_DATA");
        jumlahGagal++;
        continue;
      }

      if (String(user.statusAktif || "").trim() != "Aktif") {
        tandaiImportGagalFinal(importSheet, row, "User tidak aktif");
        jumlahGagal++;
        continue;
      }

      if (!data.nama) {
        data.nama = user.nama;
        importSheet.getRange(row, 4).setValue(data.nama);
      }

      if (!data.divisi) {
        data.divisi = user.kelasDivisi;
        importSheet.getRange(row, 5).setValue(data.divisi);
      }

      var konteksImport = buatKonteksImportOffline_(tanggalNormal, data, user);

      if (!konteksImport.valid) {
        tandaiImportGagalFinal(importSheet, row, konteksImport.pesan);
        jumlahGagal++;
        continue;
      }

      var cekAdmin = cekPrioritasKoreksiAdminUntukImport(absensiSheet, tanggalNormal, data.id);

      if (cekAdmin.terkunci) {
        tandaiImportGagalFinal(importSheet, row, cekAdmin.pesan);
        jumlahGagal++;
        continue;
      }

      var berhasil = updateAbsensiDariImportFinal(
        absensiSheet,
        tanggalNormal,
        data,
        user,
        konteksImport
      );

      if (!berhasil) {
        tandaiImportGagalFinal(importSheet, row, "Gagal update ABSENSI_HARIAN");
        jumlahGagal++;
        continue;
      }

      catatLogImportFinal(logSheet, tanggalNormal, data, user, konteksImport);

      importSheet
        .getRange(row, 11, 1, 2)
        .setValues([["Sudah Diproses", new Date()]]);

      jumlahBerhasil++;
    } catch (error) {
      tandaiImportGagalFinal(
        importSheet,
        row,
        "Error import: " + (error && error.message ? error.message : String(error))
      );
      jumlahGagal++;
    }
  }

  SpreadsheetApp.getUi().alert(
    "Import Offline selesai\n\n" +
    "Berhasil : " + jumlahBerhasil +
    "\nGagal : " + jumlahGagal +
    "\nDilewati : " + jumlahLewat
  );
}

function validasiStatusImport(status) {
  status = String(status || "").trim();

  if (!status) {
    return { valid: false, pesan: "Status_Proses kosong" };
  }

  if (status == "Belum Diproses") {
    return { valid: true, aksi: "proses" };
  }

  if (status == "Sudah Diproses" || status == "Gagal") {
    return { valid: true, aksi: "lewati" };
  }

  return { valid: false, pesan: "Status_Proses tidak valid" };
}

function validasiDataImportFinal(data) {
  if (!data.tanggal) {
    return { valid: false, pesan: "Tanggal kosong" };
  }

  if (!data.id) {
    return { valid: false, pesan: "ID kosong" };
  }

  if (!adaNilaiImport_(data.jamMasuk) && !adaNilaiImport_(data.jamPulang)) {
    return { valid: false, pesan: "Jam_Masuk dan Jam_Pulang tidak boleh sama-sama kosong" };
  }

  var tanggal = normalisasiTanggal(data.tanggal);

  if (!tanggal) {
    return { valid: false, pesan: "Tanggal tidak valid" };
  }

  return { valid: true, tanggal: tanggal };
}

function buatKonteksImportOffline_(tanggal, data, user) {
  var shiftNormal = String(user.shift || "").trim();
  var shiftEfektif = shiftNormal;

  var penugasanBackup = cariPenugasanBackupAktif_(tanggal, data.id);

  if (penugasanBackup && penugasanBackup.peran === "ASAL") {
    return {
      valid: false,
      pesan: "Import ditolak: karyawan sedang digantikan. ID Backup: " + penugasanBackup.idBackup
    };
  }

  if (penugasanBackup && penugasanBackup.peran === "PENGGANTI") {
    shiftEfektif = String(penugasanBackup.shiftBackup || "").trim();

    if (!shiftEfektif) {
      return {
        valid: false,
        pesan: "Shift_Backup kosong. ID Backup: " + penugasanBackup.idBackup
      };
    }
  } else {
    penugasanBackup = null;
  }

  var waktuMasuk = null;
  var waktuPulang = null;

  try {
    if (adaNilaiImport_(data.jamMasuk)) {
      waktuMasuk = buatWaktuAbsensiImport(tanggal, data.jamMasuk, false, null, shiftEfektif);
    }

    if (adaNilaiImport_(data.jamPulang)) {
      waktuPulang = buatWaktuAbsensiImport(tanggal, data.jamPulang, true, data.jamMasuk, shiftEfektif);
    }
  } catch (error) {
    return { valid: false, pesan: error.message };
  }

  return {
    valid: true,
    shiftNormal: shiftNormal,
    shiftEfektif: shiftEfektif,
    penugasanBackup: penugasanBackup,
    idReferensi: penugasanBackup ? String(penugasanBackup.idBackup || "").trim() : "",
    waktuMasuk: waktuMasuk,
    waktuPulang: waktuPulang
  };
}

function updateAbsensiDariImportFinal(absensiSheet, tanggal, data, user, konteksImport) {
  if (!konteksImport || !konteksImport.valid) {
    throw new Error("Konteks import tidak valid.");
  }

  var shift = konteksImport.shiftEfektif;
  var penugasanBackup = konteksImport.penugasanBackup;
  var statusKehadiran = String(data.statusKehadiran || "Hadir").trim();

  var keteranganMasuk = String(
    data.keterangan || (cekShiftFleksibel(shift) ? "Fleksibel" : "")
  ).trim();

  var adaDataYangDiproses = false;

  if (konteksImport.waktuMasuk) {
    var hasilMasuk = updateAbsensiHarian(
      absensiSheet,
      tanggal,
      konteksImport.waktuMasuk,
      data.id,
      data.nama,
      data.divisi,
      "Masuk",
      SUMBER_IMPORT_OFFLINE,
      statusKehadiran,
      keteranganMasuk,
      0,
      0,
      shift,
      penugasanBackup
    );

    if (hasilMasuk && hasilMasuk.sukses === false) {
      return false;
    }

    konteksImport.statusProsesMasuk = "Berhasil";
    adaDataYangDiproses = true;
  }

  if (konteksImport.waktuPulang) {
    var hasilPulang = updateAbsensiHarian(
      absensiSheet,
      tanggal,
      konteksImport.waktuPulang,
      data.id,
      data.nama,
      data.divisi,
      "Pulang",
      SUMBER_IMPORT_OFFLINE,
      statusKehadiran,
      data.keterangan || "",
      0,
      0,
      shift,
      penugasanBackup
    );

    if (hasilPulang && hasilPulang.sukses === false) {
      return false;
    }

    konteksImport.statusProsesPulang = (hasilPulang && hasilPulang.keterangan == "Perlu Verifikasi")
      ? "Perlu Verifikasi"
      : "Berhasil";

    adaDataYangDiproses = true;
  }

  return adaDataYangDiproses;
}

function buatWaktuAbsensiImport(tanggal, jam, isPulang, jamMasuk, shift) {
  var bagianJam = ambilBagianJamImport_(jam);
  var hasil = new Date(tanggal);

  hasil.setHours(bagianJam.jam, bagianJam.menit, bagianJam.detik, 0);

  if (!isPulang) {
    return hasil;
  }

  if (adaNilaiImport_(jamMasuk)) {
    var masuk = buatWaktuAbsensiImport(tanggal, jamMasuk, false, null, shift);

    if (hasil < masuk) {
      hasil.setDate(hasil.getDate() + 1);
    }

    return hasil;
  }

  if (shift && !cekShiftFleksibel(shift)) {
    var dataShift = bacaShiftDariTblShift(shift);

    if (dataShift.ditemukan) {
      var menitMasukShift = ubahJamKeMenit(dataShift.jamMasukShift);
      var menitPulangShift = ubahJamKeMenit(dataShift.jamPulangShift);
      var menitPulangImport = bagianJam.jam * 60 + bagianJam.menit;

      if (menitPulangShift < menitMasukShift && menitPulangImport < menitMasukShift) {
        hasil.setDate(hasil.getDate() + 1);
      }
    }
  }

  return hasil;
}

function ambilBagianJamImport_(nilaiJam) {
  if (Object.prototype.toString.call(nilaiJam) == "[object Date]") {
    if (isNaN(nilaiJam.getTime())) {
      throw new Error("Format jam tidak valid");
    }

    return {
      jam: nilaiJam.getHours(),
      menit: nilaiJam.getMinutes(),
      detik: nilaiJam.getSeconds()
    };
  }

  if (typeof nilaiJam == "number") {
    if (nilaiJam < 0 || nilaiJam >= 1) {
      throw new Error("Format jam angka harus bernilai antara 0 dan kurang dari 1");
    }

    var totalDetik = Math.round(nilaiJam * 24 * 60 * 60);
    totalDetik = totalDetik % (24 * 60 * 60);

    return {
      jam: Math.floor(totalDetik / 3600),
      menit: Math.floor((totalDetik % 3600) / 60),
      detik: totalDetik % 60
    };
  }

  var teks = String(nilaiJam || "").trim();
  var cocok = teks.match(/^([01]?\d|2[0-3]):([0-5]\d)(?::([0-5]\d))?$/);

  if (!cocok) {
    throw new Error("Format jam '" + teks + "' tidak valid. Gunakan HH:mm atau HH:mm:ss");
  }

  return {
    jam: Number(cocok[1]),
    menit: Number(cocok[2]),
    detik: Number(cocok[3] || 0)
  };
}

function adaNilaiImport_(nilai) {
  return !(
    nilai === null ||
    typeof nilai === "undefined" ||
    (typeof nilai === "string" && nilai.trim() === "")
  );
}

function catatLogImportFinal(logSheet, tanggal, data, user, konteksImport) {
  var sekarang = new Date();
  var statusKehadiran = String(data.statusKehadiran || "Hadir").trim();
  var idReferensi = konteksImport.idReferensi || "";

  var catatanSistem = idReferensi
    ? ("Import Offline sebagai karyawan pengganti. ID Backup: " + idReferensi)
    : "Import Offline";

  if (konteksImport.waktuMasuk) {
    logSheet.appendRow([
      sekarang,
      tanggal,
      konteksImport.waktuMasuk,
      data.id,
      data.nama,
      data.divisi,
      "Masuk",
      (konteksImport.statusProsesMasuk || "Berhasil"),
      SUMBER_IMPORT_OFFLINE,
      catatanSistem,
      statusKehadiran,
      0,
      0,
      idReferensi,
      ""
    ]);
  }

  if (konteksImport.waktuPulang) {
    logSheet.appendRow([
      sekarang,
      tanggal,
      konteksImport.waktuPulang,
      data.id,
      data.nama,
      data.divisi,
      "Pulang",
      (konteksImport.statusProsesPulang || "Berhasil"),
      SUMBER_IMPORT_OFFLINE,
      catatanSistem,
      statusKehadiran,
      0,
      0,
      idReferensi,
      ""
    ]);
  }
}

function tandaiImportGagalFinal(importSheet, row, pesan) {
  importSheet
    .getRange(row, 10, 1, 3)
    .setValues([[String(pesan || "Import gagal"), "Gagal", new Date()]]);
}

/****************************************************
 * API WEB APP: IMPORT OFFLINE (SUDAH DIPERBAIKI)
 ****************************************************/
function apiImportOffline(e) {
  var parameter = e && e.parameter ? e.parameter : {};
  var callback = bersihkanCallback(parameter.callback);
  var lock = LockService.getDocumentLock();

  if (!lock.tryLock(30000)) {
    return buatJsonpResponse(callback, { sukses: false, status: "Gagal", pesan: "Sistem sedang memproses data lain." });
  }

  try {
    var tanggalRaw = parameter.tanggal;
    var idKaryawan = String(parameter.id || "").trim();
    var jamMasuk = String(parameter.jam_masuk || "").trim();
    var jamPulang = String(parameter.jam_pulang || "").trim();
    var statusKehadiran = String(parameter.status_kehadiran || "Hadir").trim();
    var keterangan = String(parameter.keterangan || "").trim();

    if (!tanggalRaw || !idKaryawan) {
      return buatJsonpResponse(callback, { sukses: false, status: "Gagal", pesan: "Tanggal Kerja dan ID Karyawan wajib diisi." });
    }

    if (!jamMasuk && !jamPulang && statusKehadiran === "Hadir") {
      return buatJsonpResponse(callback, { sukses: false, status: "Gagal", pesan: "Jam Masuk atau Jam Pulang wajib diisi untuk status Hadir." });
    }

    // PERBAIKAN TAHAP 4: Menggunakan normalisasiTanggal agar aman format Indonesia (DD/MM/YYYY)
    var tanggalTugas = normalisasiTanggal(tanggalRaw);
    if (!tanggalTugas) {
      return buatJsonpResponse(callback, { sukses: false, status: "Gagal", pesan: "Format Tanggal Kerja tidak valid. Gunakan YYYY-MM-DD atau DD/MM/YYYY." });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var importSheet = ambilSheet(ss, SHEET_IMPORT_OFFLINE);
    var absensiSheet = ambilSheet(ss, SHEET_ABSENSI_HARIAN);
    var masterSheet = ambilSheet(ss, SHEET_MASTER_DATA);
    var logSheet = ambilSheet(ss, SHEET_LOG_SCAN);

    var dataKolomC = importSheet.getRange(1, 3, Math.max(importSheet.getLastRow(), 1), 1).getValues();
    var barisKosong = 2;
    for (var k = dataKolomC.length - 1; k >= 1; k--) {
      if (String(dataKolomC[k][0] || "").trim() !== "") {
        barisKosong = k + 2;
        break;
      }
    }

    var sekarang = new Date();

    var barisImportBaru = [
      sekarang,          // A: Timestamp_Input
      tanggalTugas,      // B: Tanggal
      idKaryawan,        // C: ID_Unik
      "",                // D: Nama
      "",                // E: Divisi
      jamMasuk,          // F: Jam_Masuk
      jamPulang,         // G: Jam_Pulang
      statusKehadiran,   // H: Status_Kehadiran
      "",                // I: Status_Absen
      keterangan || "Import via Web App", // J: Keterangan
      "Belum Diproses",  // K: Status_Proses
      ""                 // L: Diproses_Pada
    ];

    importSheet.getRange(barisKosong, 1, 1, barisImportBaru.length).setValues([barisImportBaru]);

    prosesImportOfflineEngineWeb_(importSheet, absensiSheet, masterSheet, logSheet);

    return buatJsonpResponse(callback, {
      sukses: true,
      status: "Berhasil",
      pesan: "Data Import Offline ID " + idKaryawan + " berhasil diproses ke ABSENSI_HARIAN."
    });

  } catch(error) {
    return buatJsonpResponse(callback, { sukses: false, status: "Error", pesan: error.message });
  } finally {
    lock.releaseLock();
  }
}

function prosesImportOfflineEngineWeb_(importSheet, absensiSheet, masterSheet, logSheet) {
  try {
    prosesImportOfflineEngine(importSheet, absensiSheet, masterSheet, logSheet);
  } catch(e) {
    if (e.message && e.message.indexOf("getUi") !== -1) {
      Logger.log("Abaikan alert UI di Web App context.");
    } else {
      throw e;
    }
  }
}

function arsipImportOffline() {
  var lock = LockService.getDocumentLock();

  if (!lock.tryLock(30000)) {
    throw new Error("Sistem sedang memproses data lain. Silakan coba kembali.");
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    arsipImportOfflineEngine(
      ambilSheet(ss, SHEET_IMPORT_OFFLINE),
      ambilSheet(ss, SHEET_ARSIP_IMPORT_OFFLINE)
    );
  } finally {
    lock.releaseLock();
  }
}

function arsipImportOfflineEngine(importSheet, arsipSheet) {
  var importData = importSheet.getDataRange().getValues();

  if (importData.length <= 1) {
    SpreadsheetApp.getUi().alert("Tidak ada data pada sheet IMPORT_OFFLINE untuk diarsipkan.");
    return;
  }

  var jumlahDiarsip = 0;

  for (var i = importData.length - 1; i >= 1; i--) {
    var row = i + 1;
    var statusProses = String(importData[i][10] || "").trim();

    if (statusProses === "Sudah Diproses" || statusProses === "Gagal") {
      var rowData = importData[i];
      arsipSheet.appendRow(rowData);
      importSheet.deleteRow(row);
      jumlahDiarsip++;
    }
  }

  SpreadsheetApp.getUi().alert(
    "Pengarsipan Import Offline Selesai\n\n" +
    "Jumlah data yang diarsipkan: " + jumlahDiarsip + " baris."
  );
}

function apiArsipImportOffline(e) {
  var parameter = e && e.parameter ? e.parameter : {};
  var callback = bersihkanCallback(parameter.callback);
  var lock = LockService.getDocumentLock();

  if (!lock.tryLock(30000)) {
    return buatJsonpResponse(callback, { sukses: false, status: "Gagal", pesan: "Sistem sedang memproses data lain." });
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var importSheet = ambilSheet(ss, SHEET_IMPORT_OFFLINE);
    var arsipSheet = ambilSheet(ss, SHEET_ARSIP_IMPORT_OFFLINE);

    arsipImportOfflineEngineWeb_(importSheet, arsipSheet);

    return buatJsonpResponse(callback, {
      sukses: true,
      status: "Berhasil",
      pesan: "Proses pengarsipan Import Offline berhasil dijalankan."
    });

  } catch(error) {
    return buatJsonpResponse(callback, { sukses: false, status: "Error", pesan: error.message });
  } finally {
    lock.releaseLock();
  }
}

function arsipImportOfflineEngineWeb_(importSheet, arsipSheet) {
